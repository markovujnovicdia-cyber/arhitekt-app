cmake_minimum_required(VERSION 3.24)
project(ArhitektLocalEngine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static private upstream libraries" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "No runtime engine plugins" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Portable architecture" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "Bounded engine threads" FORCE)
set(GGML_BLAS OFF CACHE BOOL "No external BLAS dependency" FORCE)
set(GGML_CUDA OFF CACHE BOOL "No CUDA" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "No Vulkan" FORCE)
set(GGML_RPC OFF CACHE BOOL "No network backend" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "Reproducible tool requirements" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "Portable CPU kernels" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "No additional fetched dependencies" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "Embed Metal shader source" FORCE)
if(APPLE)
    set(GGML_METAL ON CACHE BOOL "Apple Metal" FORCE)
    set(GGML_ACCELERATE ON CACHE BOOL "Apple Accelerate" FORCE)
else()
    set(GGML_METAL OFF CACHE BOOL "No Metal on Linux" FORCE)
    set(GGML_ACCELERATE OFF CACHE BOOL "No Accelerate on Linux" FORCE)
endif()
if(NOT AO_ENGINE_SOURCE OR NOT EXISTS "${AO_ENGINE_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "AO_ENGINE_SOURCE must name the pinned upstream checkout")
endif()
if(AO_ENGINE STREQUAL "whisper")
    set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(WHISPER_CURL OFF CACHE BOOL "No downloads" FORCE)
    set(WHISPER_COREML OFF CACHE BOOL "Use portable imported GGML model" FORCE)
    set(bridge AOWhisper)
    set(engine_library whisper)
    set(prefix ao_whisper_)
    set(subdirectory Whisper)
elseif(AO_ENGINE STREQUAL "llama")
    set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
    set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
    set(LLAMA_CURL OFF CACHE BOOL "No downloads" FORCE)
    set(LLAMA_HTTPLIB OFF CACHE BOOL "No HTTP" FORCE)
    set(bridge AOLlama)
    set(engine_library llama)
    set(prefix ao_llama_)
    set(subdirectory Llama)
else()
    message(FATAL_ERROR "AO_ENGINE must be whisper or llama")
endif()
add_subdirectory("${AO_ENGINE_SOURCE}" upstream EXCLUDE_FROM_ALL)
add_library(${bridge} SHARED "${subdirectory}/${bridge}.cpp")
target_include_directories(${bridge} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/${subdirectory}")
target_link_libraries(${bridge} PRIVATE ${engine_library})
set_target_properties(${bridge} PROPERTIES OUTPUT_NAME ${bridge})
if(APPLE)
    # Neither GGML version is part of the public ABI. The export whitelist is
    # essential: two separately pinned engines coexist in the same iOS process.
    target_link_options(${bridge} PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/${subdirectory}/exports.apple")
    set_target_properties(${bridge} PROPERTIES
        FRAMEWORK TRUE FRAMEWORK_VERSION A
        MACOSX_FRAMEWORK_IDENTIFIER "hr.arhitekt.office.${bridge}"
        MACOSX_FRAMEWORK_SHORT_VERSION_STRING "1.0"
        MACOSX_FRAMEWORK_BUNDLE_VERSION "1"
        PUBLIC_HEADER "${subdirectory}/${bridge}.h"
        XCODE_ATTRIBUTE_DEFINES_MODULE YES
        XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO
        INSTALL_NAME_DIR "@rpath")
    add_custom_command(TARGET ${bridge} POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_BUNDLE_DIR:${bridge}>/Modules"
        COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_CURRENT_SOURCE_DIR}/${subdirectory}/module.modulemap"
            "$<TARGET_BUNDLE_DIR:${bridge}>/Modules/module.modulemap")
else()
    target_link_options(${bridge} PRIVATE
        "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/${subdirectory}/exports.linux"
        "LINKER:--exclude-libs,ALL" "LINKER:-Bsymbolic")
    add_executable(ao_json_escape_test EXCLUDE_FROM_ALL Tests/test_json_escape.cpp)
    if(AO_ENGINE STREQUAL "llama")
        add_executable(ao_json_grammar_test EXCLUDE_FROM_ALL Tests/test_json_grammar.cpp)
        target_include_directories(ao_json_grammar_test PRIVATE "${AO_ENGINE_SOURCE}/src")
        target_link_libraries(ao_json_grammar_test PRIVATE llama)
    endif()
endif()
